Как я пишу код на Kotlin & Jetpack Compose

Пример реактивного MVI компонента с безопасным управлением состоянием и чистой структурой.

Kotlin 2.0 
@Composable
fun ProfileScreen(
    viewModel: ProfileViewModel = hiltViewModel(),
    onNavigateBack: () -> Unit
) {
    // State hoist with lifecycle-aware collection
    val state by viewModel.uiState.collectAsStateWithLifecycle()

    // One-shot side-effects (Snackbar, navigation)
    val snackbarHostState = remember { SnackbarHostState() }
    LaunchedEffect(viewModel.effect) {
        viewModel.effect.collect { effect ->
            when (effect) {
                is ProfileEffect.ShowToast -> snackbarHostState.showSnackbar(effect.message)
                ProfileEffect.NavBack -> onNavigateBack()
            }
        }
    }

    Scaffold(
        snackbarHost = { SnackbarHost(hostState = snackbarHostState) },
        topBar = { TopAppBar(title = { Text("Developer Profile") }) }
    ) { paddingValues ->
        when (state) {
            is ProfileState.Loading -> CircularProgressIndicator()
            is ProfileState.Success -> {
                ProfileContent(
                    data = state.profile,
                    onRefresh = { viewModel.handleIntent(ProfileIntent.Refresh) },
                    modifier = Modifier.padding(paddingValues)
                )
            }
            is ProfileState.Error -> ErrorStateView(msg = state.error)
        }
    }
}
@HiltViewModel
class ProfileViewModel @Inject constructor(
    private val getProfileUseCase: GetProfileUseCase
) : ViewModel() {

    private val _uiState = MutableStateFlow<ProfileState>(ProfileState.Loading)
    val uiState: StateFlow<ProfileState> = _uiState.asStateFlow()

    private val _effect = Channel<ProfileEffect>()
    val effect = _effect.receiveAsFlow()

    fun handleIntent(intent: ProfileIntent) {
        when (intent) {
            ProfileIntent.Refresh -> loadProfileData()
            is ProfileIntent.SelectProject -> onProjectClicked(intent.id)
        }
    }

    private fun loadProfileData() = viewModelScope.launch {
        _uiState.value = ProfileState.Loading
        getProfileUseCase()
            .catch { e -> _uiState.value = ProfileState.Error(e.localizedMessage) }
            .collect { profile ->
                _uiState.value = ProfileState.Success(profile)
            }
    }
}
class UserRepositoryImpl @Inject constructor(
    private val localDao: UserDao,
    private val api: ApiService,
    private val ioDispatcher: CoroutineDispatcher = Dispatchers.IO
) : UserRepository {

    // Offline-first reactive data flow
    override fun getUserProfile(): Flow<UserProfile> = flow {
        // 1. Emit cached data immediately
        val cached = localDao.getUser()
        if (cached != null) emit(cached.toDomain())

        // 2. Fetch fresh network data in background
        val remote = api.fetchProfile()
        localDao.insertUser(remote.toEntity())

        // 3. Emit fresh data
        emit(remote.toDomain())
    }.flowOn(ioDispatcher)
}
